/**
 * SignalGrid – alert router
 *
 * Routes an AlertEvent to all relevant recipients and drives delivery
 * through the retry engine. The router is responsible for:
 *
 *  1. Resolving recipients for an event (by severity + subscription rules)
 *  2. Fanning out to each recipient's preferred channel
 *  3. Delegating retries to attemptDelivery()
 *  4. Aggregating the per-recipient DeliveryRecord results
 *
 * The router deliberately does NOT implement retry logic itself — that
 * lives exclusively in retry.ts. This separation ensures the invariant
 * check in attemptDelivery() cannot be bypassed by a new routing path.
 */

import { attemptDelivery, RETRY_CONFIG } from "./retry";

/**
 * Routes an alert event to all recipients, returning one DeliveryRecord per
 * recipient. Channel selection uses the first channel in the recipient's
 * preference list.
 */
export async function routeAlert(
  event: AlertEvent,
  recipients: Recipient[] = FIXTURE_RECIPIENTS
): Promise<DeliveryRecord[]> {
  const results: DeliveryRecord[] = [];

  for (const recipient of recipients) {
    const primaryChannel = recipient.channels[0];
    const record = await attemptDelivery(
      event.eventId,
      recipient.recipientId,
      primaryChannel,
      RETRY_CONFIG
    );
    results.push(record);
  }

  return results;
}

/**
 * Returns a human-readable routing summary for display in the Memento UI.
 */
export function summariseRouting(records: DeliveryRecord[]): {
  total: number;
  delivered: number;
  failed: number;
  totalAttempts: number;
} {
  const delivered = records.filter((r) => r.finalStatus === "delivered").length;
  const totalAttempts = records.reduce((sum, r) => sum + r.attempts.length, 0);
  return {
    total: records.length,
    delivered,
    failed: records.length - delivered,
    totalAttempts,
  };
}
